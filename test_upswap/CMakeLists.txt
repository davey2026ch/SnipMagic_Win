cmake_minimum_required(VERSION 3.16)
project(TestUpSwap LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(test_upswap test_upswap.cpp)
target_include_directories(test_upswap PRIVATE ../src)
if(MSVC)
    target_compile_options(test_upswap PRIVATE /utf-8 /W3 /O2 /EHsc)
    target_compile_definitions(test_upswap PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
target_link_libraries(test_upswap PRIVATE user32 gdi32 winhttp shell32 ole32 shlwapi)
